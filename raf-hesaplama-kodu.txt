          var u = i * g[0] + j * g[1] + k * g[2];
          if (!best || u > best.u || (u === best.u && n < best.n)) { best = { u: u, n: n, a: {} }; best.a[g[0]] = i; best.a[g[1]] = j; best.a[g[2]] = k; }

    var bos = {}; g.forEach(function (w) { bos[w] = 0; }); return bos;
  }
